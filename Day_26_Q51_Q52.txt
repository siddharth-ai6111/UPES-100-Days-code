Q51. Print Right-Aligned Number Pattern
------------------------------------------------------------
CODE:
#include <stdio.h>

int main() {
    int i, j;

    for (i = 5; i >= 1; i--) {
        for (j = 1; j < i; j++)
            printf(" ");

        for (j = i; j <= 5; j++)
            printf("%d", j);

        printf("\n");
    }

    return 0;
}

SAMPLE OUTPUT:
    5
   45
  345
 2345
12345


Q52. Print 1, 3, 5, 3, 1 Stars in Vertical Form
------------------------------------------------------------
CODE:
#include <stdio.h>

int main() {
    int i, j, stars;

    for (i = 1; i <= 5; i++) {
        if (i <= 3)
            stars = 2 * i - 1;
        else
            stars = 2 * (6 - i) - 1;

        for (j = 1; j <= stars; j++)
            printf("*\n");

        if (i != 5)
            printf("\n");
    }

    return 0;
}
