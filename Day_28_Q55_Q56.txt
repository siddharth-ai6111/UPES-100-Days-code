Q55. Print All Prime Numbers from 1 to n
------------------------------------------------------------
CODE:
#include <stdio.h>

int main() {
    int n, i, j, isPrime;

    printf("Enter n: ");
    scanf("%d", &n);

    printf("Prime Numbers: ");

    for (i = 2; i <= n; i++) {
        isPrime = 1;

        for (j = 2; j * j <= i; j++) {
            if (i % j == 0) {
                isPrime = 0;
                break;
            }
        }

        if (isPrime)
            printf("%d ", i);
    }

    printf("\n");
    return 0;
}

SAMPLE OUTPUT:
Enter n: 20
Prime Numbers: 2 3 5 7 11 13 17 19


Q56. Read and Print a One-Dimensional Array
------------------------------------------------------------
CODE:
#include <stdio.h>

int main() {
    int a[100], n, i;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter elements: ");
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Array Elements: ");
    for (i = 0; i < n; i++)
        printf("%d ", a[i]);

    printf("\n");
    return 0;
}

SAMPLE OUTPUT:
Enter number of elements: 5
Enter elements: 10 20 30 40 50
Array Elements: 10 20 30 40 50
