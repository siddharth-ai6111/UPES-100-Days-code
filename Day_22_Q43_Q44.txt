Q43. Check Strong Number
------------------------------------------------------------
CODE:
#include <stdio.h>

int main() {
    int n, original, temp, digit, i, fact, sum = 0;

    printf("Enter a number: ");
    scanf("%d", &n);

    if (n < 0) {
        printf("Not a Strong Number\n");
        return 0;
    }

    original = n;
    temp = n;

    do {
        digit = temp % 10;
        fact = 1;

        for (i = 1; i <= digit; i++)
            fact = fact * i;

        sum = sum + fact;
        temp = temp / 10;
    } while (temp != 0);

    if (sum == original)
        printf("Strong Number\n");
    else
        printf("Not a Strong Number\n");

    return 0;
}

SAMPLE OUTPUT:
Enter a number: 145
Strong Number


Q44. Sum of Series: 1 + 3/4 + 5/6 + 7/8 + ... up to n Terms
------------------------------------------------------------
CODE:
#include <stdio.h>

int main() {
    int n, i;
    float sum = 0;

    printf("Enter number of terms: ");
    scanf("%d", &n);

    for (i = 1; i <= n; i++) {
        if (i == 1)
            sum = sum + 1.0;
        else
            sum = sum + (float)(2 * i - 1) / (2 * i);
    }

    printf("Sum = %.4f\n", sum);
    return 0;
}

SAMPLE OUTPUT:
Enter number of terms: 4
Sum = 3.4583
