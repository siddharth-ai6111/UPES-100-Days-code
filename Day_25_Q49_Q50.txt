Q49. Print Reverse Starting Number Pattern
------------------------------------------------------------
CODE:
#include <stdio.h>

int main() {
    int i, j;

    for (i = 5; i >= 1; i--) {
        for (j = i; j <= 5; j++)
            printf("%d", j);
        printf("\n");
    }

    return 0;
}

SAMPLE OUTPUT:
5
45
345
2345
12345


Q50. Print Right Shifted Decreasing Star Pattern
------------------------------------------------------------
CODE:
#include <stdio.h>

int main() {
    int i, j;

    for (i = 0; i < 5; i++) {
        for (j = 0; j < i; j++)
            printf(" ");

        for (j = 0; j < 5 - i; j++)
            printf("*");

        printf("\n");
    }

    return 0;
}

SAMPLE OUTPUT:
*****
 ****
  ***
   **
    *
