Q11. Check Even or Odd
------------------------------------------------------------
CODE:
#include <stdio.h>

int main() {
    int n;

    printf("Enter an integer: ");
    scanf("%d", &n);

    if (n % 2 == 0)
        printf("Even\n");
    else
        printf("Odd\n");

    return 0;
}

SAMPLE OUTPUT:
Enter an integer: 17
Odd


Q12. Check Positive, Negative or Zero using Nested If-Else
------------------------------------------------------------
CODE:
#include <stdio.h>

int main() {
    int n;

    printf("Enter an integer: ");
    scanf("%d", &n);

    if (n >= 0) {
        if (n == 0)
            printf("Zero\n");
        else
            printf("Positive\n");
    } else {
        printf("Negative\n");
    }

    return 0;
}

SAMPLE OUTPUT:
Enter an integer: -8
Negative
