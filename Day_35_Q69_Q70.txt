Q69. Find Second Largest Element in an Array
------------------------------------------------------------
CODE:
#include <stdio.h>

int main() {
    int a[100], n, i;
    int largest, second, found = 0;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    if (n < 2) {
        printf("At least two elements are required\n");
        return 0;
    }

    printf("Enter elements: ");
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    largest = a[0];

    for (i = 1; i < n; i++) {
        if (a[i] > largest)
            largest = a[i];
    }

    for (i = 0; i < n; i++) {
        if (a[i] != largest) {
            second = a[i];
            found = 1;
            break;
        }
    }

    if (!found) {
        printf("No second largest distinct element\n");
        return 0;
    }

    for (i = 0; i < n; i++) {
        if (a[i] != largest && a[i] > second)
            second = a[i];
    }

    printf("Second Largest = %d\n", second);
    return 0;
}

SAMPLE OUTPUT:
Enter number of elements: 5
Enter elements: 10 50 30 40 20
Second Largest = 40


Q70. Rotate an Array Right by k Positions
------------------------------------------------------------
CODE:
#include <stdio.h>

int main() {
    int a[100], n, k, i, j, last;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    if (n <= 0) {
        printf("Invalid size\n");
        return 0;
    }

    printf("Enter elements: ");
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Enter k: ");
    scanf("%d", &k);

    k = k % n;

    for (i = 1; i <= k; i++) {
        last = a[n - 1];

        for (j = n - 1; j > 0; j--)
            a[j] = a[j - 1];

        a[0] = last;
    }

    printf("Array after rotation: ");
    for (i = 0; i < n; i++)
        printf("%d ", a[i]);

    printf("\n");
    return 0;
}

SAMPLE OUTPUT:
Enter number of elements: 5
Enter elements: 1 2 3 4 5
Enter k: 2
Array after rotation: 4 5 1 2 3
